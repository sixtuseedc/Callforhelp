    <script>
        // Default Configuration
        const DEFAULT_PHONE = "2348000000000"; // Default placeholder Nigerian phone number
        const DEFAULT_MSG = "Hello Callforhelp, I am reaching out to talk to someone.";

        // DOM Elements
        const modal = document.getElementById('settings-modal');
        const openBtn = document.getElementById('open-settings-btn');
        const footerOpenBtn = document.getElementById('footer-settings-link');
        const closeBtn = document.getElementById('close-settings-btn');
        const cancelBtn = document.getElementById('cancel-settings-btn');
        const form = document.getElementById('phone-config-form');
        const phoneInput = document.getElementById('whatsapp-number');
        const msgInput = document.getElementById('greeting-msg');
        const displayPhoneNum = document.getElementById('display-phone-num');

        // CTA Links
        const heroCta = document.getElementById('hero-main-cta');
        const headerCta = document.getElementById('header-whatsapp-btn');
        const secondaryCta = document.getElementById('secondary-whatsapp-btn');

        // Initialize icons
        lucide.createIcons();

        // Set Current Year
        document.getElementById('year').textContent = new Date().getFullYear();

        // Function to Format Phone for Display
        function formatPhoneDisplay(phone) {
            if (!phone) return '+234 800 000 0000';
            if (phone.startsWith('234')) {
                return `+234 ${phone.slice(3, 6)} ${phone.slice(6, 9)} ${phone.slice(9)}`;
            }
            return `+${phone}`;
        }

        // Load Settings from LocalStorage
        function loadSettings() {
            const savedPhone = localStorage.getItem('callforhelp_phone') || DEFAULT_PHONE;
            const savedMsg = localStorage.getItem('callforhelp_msg') || DEFAULT_MSG;

            phoneInput.value = savedPhone;
            msgInput.value = savedMsg;
            displayPhoneNum.textContent = formatPhoneDisplay(savedPhone);

            // Construct WhatsApp Link
            const encodedMsg = encodeURIComponent(savedMsg);
            const waUrl = `https://wa.me/${savedPhone}?text=${encodedMsg}`;

            // Apply links to CTAs
            heroCta.setAttribute('href', waUrl);
            headerCta.setAttribute('href', waUrl);
            secondaryCta.setAttribute('href', waUrl);
        }

        // Modal Controls
        function openModal() {
            modal.classList.remove('hidden');
        }

        function closeModal() {
            modal.classList.add('hidden');
        }

        openBtn.addEventListener('click', openModal);
        footerOpenBtn.addEventListener('click', openModal);
        closeBtn.addEventListener('click', closeModal);
        cancelBtn.addEventListener('click', closeModal);

        // Save Settings
        form.addEventListener('submit', (e) => {
            e.preventDefault();
            const cleanPhone = phoneInput.value.replace(/[^0-9]/g, '');
            const msg = msgInput.value.trim();

            if (cleanPhone) {
                localStorage.setItem('callforhelp_phone', cleanPhone);
                localStorage.setItem('callforhelp_msg', msg);
                loadSettings();
                closeModal();
            }
        });

        // Close Modal on backdrop click
        window.addEventListener('click', (e) => {
            if (e.target === modal) {
                closeModal();
            }
        });

        // Initial Load
        loadSettings();
    </script>
